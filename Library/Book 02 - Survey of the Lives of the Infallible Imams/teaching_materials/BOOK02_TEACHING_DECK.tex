\begin{frame}{The Puzzle: Why Do the Imams Look Different?}
\bigquote{Outward method must be read in its circumstances.}{Teaching synthesis}
\begin{itemize}
\item al-Hassan — peace
\item al-Husayn — revolt
\item Zayn al-'Abidin — worship, service, mourning
\item al-Sadiq — scholarship and political restraint
\item Musa al-Kazim — endurance under imprisonment
\item al-Rida — crown-prince appointment
\item al-'Askari — leadership under surveillance
\end{itemize}
\end{frame}

\begin{frame}{The Interpretive Key}
\begin{itemize}
\item enduring principles can remain stable
\item outward forms can respond to changing conditions
\item inherited custom is not automatically a permanent command
\item apparent contradictions require contextual reconciliation
\end{itemize}
\end{frame}

\begin{frame}{'Ali: Justice Has a Cost}
\bigquote{Political success is not worth purchasing with oppression.}{Chapter 1 teaching synthesis}
\begin{itemize}
\item rejects special advantage for influential people
\item refuses to trade the rights of the weak for political ease
\item rejects deception as a moral political method
\end{itemize}
\end{frame}

\begin{frame}{The Kharijite Problem}
\begin{itemize}
\item impressive outward worship
\item rigid slogans and judgments
\item shallow understanding
\item hostility toward the Imam
\end{itemize}
\bigquote{Investigate before you judge.}{Lesson of the Iqbal / Ja'far-Sadiq anecdote}
\end{frame}

\begin{frame}{War or Peace?}
\begin{itemize}
\item the Prophet used restraint, warfare and treaty in different circumstances
\item jihad has different legal categories
\item a truce can serve Muslim interests
\item peace is not automatically weakness
\end{itemize}
\end{frame}

\begin{frame}{al-Hassan vs al-Husayn}
\centering\scriptsize
\renewcommand{\arraystretch}{1.35}
\begin{tabular}{|l|p{4.5cm}|p{4.5cm}|}\hline
&\textbf{al-Hassan}&\textbf{al-Husayn}\\\hline
Position&sitting caliph&no governing office\\\hline
Kufah&exhausted / demoralized&mass invitation\\\hline
Allegiance&not demanded&demanded\\\hline
Military situation&unfavorable&different later conditions\\\hline
Political record&future tyranny not fully exposed&oppression clearer\\\hline
\end{tabular}
\end{frame}

\begin{frame}{Zayn al-'Abidin: Resistance Through Service}
\begin{itemize}
\item deep absorption in worship
\item kindness to isolated and poor people
\item serving strangers during Hajj
\item preserving Karbala's message through remembrance
\end{itemize}
\end{frame}

\begin{frame}{Musa al-Kazim: Dignity Under Pressure}
\begin{itemize}
\item years of imprisonment
\item repeated attempts to obtain submission
\item refusal to legitimize Harun
\item spiritual influence despite confinement
\end{itemize}
\end{frame}

\begin{frame}{al-Sadiq: An Opening Is Not Automatically a Calling}
\begin{itemize}
\item Umayyad collapse creates a political opening
\item Abbasid organizers use broad Islamic slogans
\item Abu Muslim and Abu Salmah represent organized political action
\item political slogans do not equal reliable conditions
\end{itemize}
\end{frame}

\begin{frame}{al-Rida: Appointment Is Not Endorsement}
\begin{itemize}
\item Ma'mun initiates the proposal
\item al-Rida initially refuses
\item acceptance occurs under pressure
\item non-interference condition
\item independent conduct remains visible
\end{itemize}
\bigquote{Separate facts from disputed motives.}{Methodological rule of Chapter 6}
\end{frame}

\begin{frame}{al-'Askari: Imamate Under Surveillance}
\begin{itemize}
\item Samarra and military surroundings
\item intense monitoring
\item searches for the expected child
\item concealed birth
\item household protection and Jaddah
\end{itemize}
\end{frame}

\begin{frame}{Universal Justice}
\bigquote{Justice belongs to the general prophetic mission.}{Teaching synthesis of Chapter 8 Part 1}
\begin{itemize}
\item worldly justice matters
\item justice is not simply identical treatment
\item human moral capacity matters
\item 'Ali gives a concrete example
\end{itemize}
\end{frame}

\begin{frame}{Mahdism: History, Evidence, Claims}
\begin{itemize}
\item belief predates the third century AH
\item Qur'anic foundations are discussed
\item prophetic reports are discussed
\item Sunni and Shi'ah narrations enter the historical argument
\item history also contains false claimants and political appropriation
\end{itemize}
\end{frame}

\begin{frame}{Awaiting Is Not Passivity}
\bigquote{Awaiting does not invalidate obligatory duties.}{Mutahhari, Chapter 8 Part 2}
\begin{itemize}
\item do not abandon prayer or fasting
\item do not abandon moral education
\item do not abandon reform
\item do not stop enjoining good and forbidding evil
\end{itemize}
\end{frame}
